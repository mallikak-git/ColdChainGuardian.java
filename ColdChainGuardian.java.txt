    // ==========================================
    // TEMPERATURE STATUS CHECK
    // ==========================================

    private static String checkTemperature(
            double temperature) {

        if (temperature < MIN_SAFE_TEMPERATURE ||
                temperature > MAX_SAFE_TEMPERATURE) {

            return "BREACH";

        } else if (
                temperature <=
                        MIN_SAFE_TEMPERATURE
                                + WARNING_MARGIN
                        ||
                        temperature >=
                                MAX_SAFE_TEMPERATURE
                                        - WARNING_MARGIN) {

            return "WARNING";

        } else {

            return "SAFE";
        }
    }


    // ==========================================
    // PROACTIVE BREACH PREDICTION
    // ==========================================

    private static String predictTemperatureRisk(
            double currentTemperature,
            double previousTemperature) {

        double temperatureChange =
                currentTemperature
                        - previousTemperature;


        // Rising temperature approaching
        // upper safe limit

        if (temperatureChange > 0 &&
                currentTemperature >= 7.0 &&
                currentTemperature <
                        MAX_SAFE_TEMPERATURE) {

            return "PREDICTED_BREACH";
        }


        // Falling temperature approaching
        // lower safe limit

        if (temperatureChange < 0 &&
                currentTemperature <= 3.0 &&
                currentTemperature >
                        MIN_SAFE_TEMPERATURE) {

            return "PREDICTED_BREACH";
        }


        return "NO_IMMEDIATE_RISK";
    }


    // ==========================================
    // RISK ANALYZER
    // ==========================================

    private static void analyzeTemperature(
            double currentTemperature,
            double previousTemperature) {


        totalReadings++;


        String status =
                checkTemperature(
                        currentTemperature
                );


        if ("WARNING".equals(status)) {

            warningCount++;
        }


        String prediction =
                "NO_IMMEDIATE_RISK";


        if (previousTemperature !=
                currentTemperature) {

            prediction =
                    predictTemperatureRisk(
                            currentTemperature,
                            previousTemperature
                    );
        }


        if ("PREDICTED_BREACH".equals(
                prediction)) {

            predictedBreachCount++;
        }


        if ("BREACH".equals(status)) {

            actualBreachCount++;
        }


        System.out.println(
                "Risk Analyzer -> "
                        + "Temperature: "
                        + currentTemperature
                        + " °C"
                        + " | Status: "
                        + status
                        + " | Prediction: "
                        + prediction
        );


        if ("PREDICTED_BREACH".equals(
                prediction)) {

            System.out.println(
                    "PROACTIVE ALERT: "
                            + "Temperature is moving "
                            + "toward an unsafe level!"
            );
        }


        if ("BREACH".equals(status)) {

            System.out.println(
                    "CRITICAL ALERT: "
                            + "Cold-chain temperature "
                            + "breach detected!"
            );
        }
    }


    // ==========================================
    // REACTIVE VS PROACTIVE COMPARISON
    // ==========================================

    private static void compareReactiveAndProactive(
            double[] temperatureReadings) {


        int reactiveReading = -1;

        int proactiveReading = -1;


        for (int i = 1;
             i < temperatureReadings.length;
             i++) {


            double current =
                    temperatureReadings[i];


            double previous =
                    temperatureReadings[i - 1];


            // ==========================================
            // REACTIVE SYSTEM
            // ==========================================

            if (reactiveReading == -1 &&
                    current > MAX_SAFE_TEMPERATURE) {

                reactiveReading = i;
            }


            // ==========================================
            // PROACTIVE SYSTEM
            // ==========================================

            String prediction =
                    predictTemperatureRisk(
                            current,
                            previous
                    );


            if (proactiveReading == -1 &&
                    "PREDICTED_BREACH".equals(
                            prediction)) {

                proactiveReading = i;
            }
        }


        reactiveAlertReading =
                reactiveReading;


        proactiveAlertReading =
                proactiveReading;


        System.out.println();
        System.out.println(
                "======================================"
        );
        System.out.println(
                "   REACTIVE vs PROACTIVE"
        );
        System.out.println(
                "======================================"
        );


        System.out.println(
                "Reactive alert reading = "
                        + reactiveReading
        );


        System.out.println(
                "Proactive alert reading = "
                        + proactiveReading
        );


        if (reactiveReading != -1 &&
                proactiveReading != -1) {

            int readingsEarly =
                    reactiveReading
                            - proactiveReading;


            System.out.println(
                    "Early warning advantage = "
                            + readingsEarly
                            + " reading(s)"
            );
        }


        System.out.println(
                "======================================"
        );
    }


    // ==========================================
    // RESULT DATASET
    // ==========================================

    private static void printResultDataset(
            double[] temperatureReadings) {


        System.out.println();

        System.out.println(
                "=============================================================="
        );

        System.out.println(
                "                 COLD CHAIN RESULT DATASET"
        );

        System.out.println(
                "=============================================================="
        );


        System.out.printf(
                "%-8s %-15s %-15s %-20s%n",
                "Reading",
                "Temperature",
                "Status",
                "Prediction"
        );


        System.out.println(
                "--------------------------------------------------------------"
        );


        double previousTemperature =
                temperatureReadings[0];


        for (int i = 0;
             i < temperatureReadings.length;
             i++) {


            double currentTemperature =
                    temperatureReadings[i];


            String status =
                    checkTemperature(
                            currentTemperature
                    );


            String prediction =
                    "NO_IMMEDIATE_RISK";


            if (i > 0) {

                prediction =
                        predictTemperatureRisk(
                                currentTemperature,
                                previousTemperature
                        );
            }


            System.out.printf(
                    "%-8d %-15.1f %-15s %-20s%n",
                    i + 1,
                    currentTemperature,
                    status,
                    prediction
            );


            previousTemperature =
                    currentTemperature;
        }


        System.out.println(
                "=============================================================="
        );
    }


    // ==========================================
    // CREATE FOG DEVICE
    // ==========================================

    private static FogDevice createFogDevice(
            String nodeName,
            long mips,
            int ram,
            long upBw,
            long downBw,
            int level,
            double ratePerMips,
            double busyPower,
            double idlePower) {


        List<Pe> peList =
                new ArrayList<Pe>();


        peList.add(
                new Pe(
                        0,
                        new PeProvisionerOverbooking(
                                mips
                        )
                )
        );


        int hostId =
                FogUtils.generateEntityId();


        long storage =
                1000000;


        int bw =
                10000;


        PowerHost host =
                new PowerHost(
                        hostId,
                        new RamProvisionerSimple(
                                ram
                        ),
                        new BwProvisionerOverbooking(
                                bw
                        ),
                        storage,
                        peList,
                        new StreamOperatorScheduler(
                                peList
                        ),
                        new FogLinearPowerModel(
                                busyPower,
                                idlePower
                        )
                );


        List<Host> hostList =
                new ArrayList<Host>();


        hostList.add(host);


        String arch =
                "x86";

        String os =
                "Linux";

        String vmm =
                "Xen";


        double time_zone =
                10.0;

        double cost =
                3.0;

        double costPerMem =
                0.05;

        double costPerStorage =
                0.001;

        double costPerBw =
                0.0;


        LinkedList<Storage> storageList =
                new LinkedList<Storage>();


        FogDeviceCharacteristics characteristics =
                new FogDeviceCharacteristics(
                        arch,
                        os,
                        vmm,
                        host,
                        time_zone,
                        cost,
                        costPerMem,
                        costPerStorage,
                        costPerBw
                );


        FogDevice fogdevice =
                null;


        try {

            fogdevice =
                    new FogDevice(
                            nodeName,
                            characteristics,
                            new AppModuleAllocationPolicy(
                                    hostList
                            ),
                            storageList,
                            10,
                            upBw,
                            downBw,
                            0,
                            ratePerMips
                    );

        } catch (Exception e) {

            e.printStackTrace();

        }


        fogdevice.setLevel(level);


        return fogdevice;
    }


    // ==========================================
    // CREATE COLD CHAIN APPLICATION
    // ==========================================

    @SuppressWarnings({"serial"})
    private static Application createColdChainApplication(
            String appId,
            int userId) {


        Application application =
                Application.createApplication(
                        appId,
                        userId
                );


        // ==========================================
        // APPLICATION MODULES
        // ==========================================

        application.addAppModule(
                "temperature_monitor",
                10
        );


        application.addAppModule(
                "risk_analyzer",
                10
        );


        application.addAppModule(
                "alert_system",
                10
        );


        // ==========================================
        // SENSOR → TEMPERATURE MONITOR
        // ==========================================

        application.addAppEdge(
                "TEMP",
                "temperature_monitor",
                1000,
                1000,
                "TEMP_DATA",
                Tuple.UP,
                AppEdge.SENSOR
        );


        // ==========================================
        // TEMPERATURE MONITOR → RISK ANALYZER
        // ==========================================

        application.addAppEdge(
                "temperature_monitor",
                "risk_analyzer",
                500,
                500,
                "TEMPERATURE_READING",
                Tuple.UP,
                AppEdge.MODULE
        );


        // ==========================================
        // RISK ANALYZER → ALERT SYSTEM
        // ==========================================

        application.addAppEdge(
                "risk_analyzer",
                "alert_system",
                500,
                500,
                "RISK_ALERT",
                Tuple.UP,
                AppEdge.MODULE
        );


        // ==========================================
        // SENSOR DATA MAPPING
        // ==========================================

        application.addTupleMapping(
                "temperature_monitor",
                "TEMP_DATA",
                "TEMPERATURE_READING",
                new FractionalSelectivity(1.0)
        );


        // ==========================================
        // RISK ALERT MAPPING
        // ==========================================

        application.addTupleMapping(
                "risk_analyzer",
                "TEMPERATURE_READING",
                "RISK_ALERT",
                new FractionalSelectivity(1.0)
        );


        return application;
    }


    // ==========================================
    // SAVE TEMPERATURE RESULTS TO CSV
    // ==========================================

    private static void saveResultsToCSV(
            double[] temperatureReadings) {

        String fileName = "cold_chain_results.csv";

        try {
            // Train both standalone Weka models once for this CSV export.
            ColdChainML.TrainedModel mlModel = ColdChainML.trainModel();
            FutureBreachML.Predictor futureModel = FutureBreachML.trainModel();

            int currentRiskEvaluated = 0;
            int currentRiskCorrectCount = 0;
            int futureBreachEvaluated = 0;
            int futureBreachCorrect = 0;

            try (PrintWriter writer = new PrintWriter(new FileWriter(fileName))) {
                writer.println("Reading,Temperature,Status,RuleBasedPrediction,MLPrediction,FutureBreachMLPrediction,ActualFutureBreach,CurrentRiskMLCorrect,FutureMLCorrect");

                double previousTemperature = temperatureReadings.length > 0
                        ? temperatureReadings[0] : Double.NaN;

                for (int i = 0; i < temperatureReadings.length; i++) {
                    double currentTemperature = temperatureReadings[i];
                    String status = checkTemperature(currentTemperature);
                    String ruleBasedPrediction = "NO_IMMEDIATE_RISK";
                    String mlPrediction = "NOT_AVAILABLE";
                    String futureBreachPrediction = "NOT_AVAILABLE";
                    String actualFutureBreach = "NOT_AVAILABLE";
                    String currentRiskCorrect = "NOT_AVAILABLE";
                    String futureMLCorrect = "NOT_AVAILABLE";

                    if (i > 0) {
                        ruleBasedPrediction = predictTemperatureRisk(
                                currentTemperature, previousTemperature);

                        try {
                            mlPrediction = mlModel.predictRisk(
                                    currentTemperature, previousTemperature);
                            currentRiskCorrect = mlPrediction.equals(status) ? "YES" : "NO";
                            currentRiskEvaluated++;
                            if ("YES".equals(currentRiskCorrect)) currentRiskCorrectCount++;
                        } catch (Exception e) {
                            mlPrediction = "ML_ERROR";
                            currentRiskCorrect = "NOT_AVAILABLE";
                            System.out.println("Current-risk ML prediction failed at reading "
                                    + (i + 1) + ": " + e.getMessage());
                        }

                        try {
                            futureBreachPrediction = futureModel.predictFutureBreach(
                                    currentTemperature, previousTemperature);
                        } catch (Exception e) {
                            futureBreachPrediction = "ML_ERROR";
                            System.out.println("Future-breach ML prediction failed at reading "
                                    + (i + 1) + ": " + e.getMessage());
                        }
                    }

                    // Evaluate only when TWO later readings are available.
                    if (i + 2 < temperatureReadings.length) {
                        boolean breachInNextTwo = false;
                        for (int j = i + 1; j <= i + 2; j++) {
                            if (temperatureReadings[j] < MIN_SAFE_TEMPERATURE
                                    || temperatureReadings[j] > MAX_SAFE_TEMPERATURE) {
                                breachInNextTwo = true;
                                break;
                            }
                        }
                        actualFutureBreach = breachInNextTwo
                                ? "BREACH_EXPECTED" : "NO_BREACH_EXPECTED";

                        if (!"NOT_AVAILABLE".equals(futureBreachPrediction)
                                && !"ML_ERROR".equals(futureBreachPrediction)) {
                            futureMLCorrect = futureBreachPrediction.equals(actualFutureBreach)
                                    ? "YES" : "NO";
                            futureBreachEvaluated++;
                            if ("YES".equals(futureMLCorrect)) futureBreachCorrect++;
                        }
                    }

                    writer.println((i + 1) + "," + currentTemperature + "," + status + ","
                            + ruleBasedPrediction + "," + mlPrediction + ","
                            + futureBreachPrediction + "," + actualFutureBreach + ","
                            + currentRiskCorrect + "," + futureMLCorrect);

                    previousTemperature = currentTemperature;
                }
            }

            System.out.println();
            System.out.println("CSV RESULTS SAVED SUCCESSFULLY: " + fileName);
            System.out.println("Current-risk ML correctness: " + currentRiskCorrectCount
                    + "/" + currentRiskEvaluated + " evaluated readings");
            System.out.println("Future-breach ML correctness: " + futureBreachCorrect
                    + "/" + futureBreachEvaluated + " evaluated readings");
            System.out.println("Note: Future-breach validation is marked NOT_AVAILABLE when fewer than two future readings exist.");
            System.out.println("Note: This is a small, predefined temperature sequence; these counts are not real-world accuracy metrics.");

        } catch (Exception e) {
            System.out.println("Error while saving CSV:");
            e.printStackTrace();
        }
    }
